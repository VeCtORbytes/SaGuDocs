## Interfaces

```ts
interface SessionStore {
  getHistory(sessionId: string): Promise<Message[]>;
  appendMessages(sessionId: string, messages: Message[]): Promise<void>;
  clear(sessionId: string): Promise<void>;
}

interface GraphMemoryEntry {
  subject: string;
  predicate: string;
  object: string;
  confidence?: number;
  mentions?: number;
  source?: string;
  createdAt?: number;
  updatedAt?: number;
  metadata?: Record<string, unknown>;
}
interface FactKey { subject: string; predicate: string; object: string; }
interface GraphMemoryQuery { about: string; limit?: number; }

interface GraphMemoryStore {
  remember(entry: GraphMemoryEntry): Promise<void>;
  recall(query: GraphMemoryQuery): Promise<GraphMemoryEntry[]>;
  listFacts(): Promise<GraphMemoryEntry[]>;
  forget(key: FactKey): Promise<void>;
  mergeEntities(from: string, into: string): Promise<void>;
  close(): Promise<void>;
}
```

## Graph memory tools

```ts
function createGraphMemoryTools(store: GraphMemoryStore): Tool[];
// [remember_fact, recall_facts]
```

| Tool | Input | Returns |
|---|---|---|
| `remember_fact` | `{ subject: string; predicate: string; object: string }` | `{ stored: true, fact, message }` |
| `recall_facts` | `{ about: string; limit?: number /* default 5 */ }` | `{ count, facts: GraphMemoryEntry[], summary }` |

## Long-term memory

```ts
new LongTermMemory(options: LongTermMemoryOptions)

class LongTermMemory implements LongTermMemoryHook {
  readonly store: GraphMemoryStore;
  readonly extractor: FactExtractor;
  readonly relationshipBuilder: RelationshipBuilder;
  readonly maintainer: GraphMaintainer;
```

## Background processes

```ts
abstract class GraphProcess {
  readonly name: GraphProcessName;          // "fact_extractor" | "relationship_builder" | "graph_maintainer"
  readonly store: GraphMemoryStore;
  readonly intervalMs: number;
  get isRunning(): boolean;
  start(): void;
  stop(): Promise<void>;
  runOnce(): Promise<number>;               // changes made; never throws
  protected abstract work(ctx: ProcessContext): Promise<number>;
}

interface GraphProcessOptions { store; intervalMs?; maxRetries?; onEvent?: GraphEventListener; traceExporter? }

new FactExtractor(options: GraphProcessOptions & { model; minConfidence?; batchSize?; maxAttempts?; instructions? })
  // enqueue(input: { messages: Message[]; sessionId?: string }): void;  get pending(): number
new RelationshipBuilder(options: GraphProcessOptions & { rules?; symmetricPredicates?; minConfidence? })
new GraphMaintainer(options: GraphProcessOptions & { halfLifeMs?; pruneBelow?; singleValuedPredicates? })

interface InferenceRule { name: string; when: [string, string]; then: string; }
const DEFAULT_INFERENCE_RULES: InferenceRule[];
const DEFAULT_SYMMETRIC_PREDICATES: string[];
const DEFAULT_SINGLE_VALUED_PREDICATES: string[];
```

Extend `GraphProcess` and implement `work(ctx)` to add your own process. Call `ctx.record(update)` for every change.

## Graph events

```ts
type GraphUpdate =
  | { operation: "fact_added" | "fact_reinforced"; fact: GraphMemoryEntry }
  | { operation: "relationship_inferred" | "relationship_updated"; fact: GraphMemoryEntry; rule: string }
  | { operation: "entities_merged"; from: string; into: string }
  | { operation: "conflict_resolved"; kept: GraphMemoryEntry; removed: GraphMemoryEntry }
  | { operation: "fact_pruned"; fact: GraphMemoryEntry; reason: "stale" | "low_confidence"; effectiveConfidence: number };

type GraphEvent =
  | ({ type: "graph_updated"; process: GraphProcessName } & GraphUpdate)
  | { type: "process_completed"; process: GraphProcessName; changes: number; durationMs: number; traceId: string }
  | { type: "process_failed"; process: GraphProcessName; error: unknown; attempt: number; willRetry: boolean };
```
