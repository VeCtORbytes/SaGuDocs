## Guardrail types

```ts
interface GuardrailResult<T = string> { pass: boolean; reason?: string; modified?: T; }
interface GuardrailContext { agentName?: string; runId?: string; [key: string]: unknown; }

type InputGuardrail  = (input: string,     context?: GuardrailContext) => GuardrailResult<string>   | Promise<GuardrailResult<string>>;
type OutputGuardrail = (output: string,    context?: GuardrailContext) => GuardrailResult<string>   | Promise<GuardrailResult<string>>;
type ToolGuardrail   = (toolCall: ToolCall, context?: GuardrailContext) => GuardrailResult<ToolCall> | Promise<GuardrailResult<ToolCall>>;
```

The runner ignores `modified` from tool guardrails; only `pass` and `reason` are used for tools.

## Guardrail helpers

```ts
function createRegexInputGuardrail(options: { pattern: RegExp; reason?: string; replaceWith?: string }): InputGuardrail;
function createRegexOutputGuardrail(options: { pattern: RegExp; reason?: string; replaceWith?: string }): OutputGuardrail;
function createLengthGuardrail(options: { minLength?: number; maxLength?: number; stage?: "input" | "output" }): (text: string) => GuardrailResult<string>;
function createToolAllowlistGuardrail(allowedToolNames: string[]): ToolGuardrail;
function createToolParameterGuardrail(
  toolName: string,
  validator: (args: Record<string, unknown> | string) => boolean | Promise<boolean>,
  reason?: string,
): ToolGuardrail;
```
